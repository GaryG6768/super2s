<script>
const SUPABASE_URL="https://hwgxfgmmupnxhtviuqks.supabase.co";
const SUPABASE_KEY="sb_publishable_4MAxToZrtx1OUoLj5s6d1g_TC8rHqrM";
const sb=supabase.createClient(SUPABASE_URL,SUPABASE_KEY);

let currentWeek=null,playerId=null,currentGames=[],wheelGames=[],spinCount=0,spinning=false,wheelRotation=0;
const $=id=>document.getElementById(id);
function esc(s){return String(s??"").replace(/[&<>"']/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#039;"}[c]))}
function msg(t){$("message").innerHTML='<div class="msg">'+esc(t)+"</div>"}
function dateLabel(d){return new Intl.DateTimeFormat("en-GB",{weekday:"short",day:"numeric",month:"short",hour:"2-digit",minute:"2-digit"}).format(new Date(d))}

function drawWheel(){
  const list=wheelGames,canvas=$("wheelCanvas"),stage=$("wheelStage");
  if(!canvas||!stage||!list.length)return;
  const n=list.length,size=Math.max(300,Math.min(430,Math.floor(innerWidth*.94)));
  stage.style.width=size+"px";stage.style.height=size+"px";
  canvas.width=size*2;canvas.height=size*2;
  const ctx=canvas.getContext("2d"),r=size,cx=r,cy=r,seg=Math.PI*2/n;
  ctx.clearRect(0,0,size*2,size*2);
  const colours=["#0b1d35","#12345a","#0b2442","#173e67","#0a1a31","#16436f"];

  list.forEach((g,i)=>{
    const a0=-Math.PI/2+i*seg+.008,a1=-Math.PI/2+(i+1)*seg-.008,mid=(a0+a1)/2;
    const g1=spinCount>=1&&currentGames[0]&&g.home===currentGames[0].home&&g.away===currentGames[0].away;
    const g2=spinCount>=2&&currentGames[1]&&g.home===currentGames[1].home&&g.away===currentGames[1].away;
    ctx.beginPath();ctx.moveTo(cx,cy);ctx.arc(cx,cy,r-5,a0,a1);ctx.closePath();
    ctx.fillStyle=(g1||g2)?"#ffd400":colours[i%colours.length];ctx.fill();
    ctx.strokeStyle=(g1||g2)?"#fff":"#8ea4bd";ctx.lineWidth=(g1||g2)?4:2;ctx.stroke();

    const fs=Math.max(8,Math.min(15,210/n));
    ctx.save();ctx.translate(cx+Math.cos(mid)*r*.70,cy+Math.sin(mid)*r*.70);
    ctx.rotate(mid+Math.PI/2);ctx.textAlign="center";ctx.textBaseline="middle";
    ctx.fillStyle=(g1||g2)?"#07111f":"#fff";ctx.font=`900 ${fs}px Arial`;
    let label=g.home+" v "+g.away;if(label.length>30)label=label.slice(0,29)+"…";
    ctx.fillText(label,0,-5);
    ctx.fillStyle=(g1||g2)?"#07111f":"#39a1ff";ctx.font=`900 ${Math.max(8,fs-1)}px Arial`;
    ctx.fillText(dateLabel(g.kick).replace(/^[A-Za-z]{3},?\s*/,""),0,fs+7);
    ctx.restore();
  });
  ctx.beginPath();ctx.arc(cx,cy,r-7,0,Math.PI*2);ctx.strokeStyle="#1785ff";ctx.lineWidth=3;ctx.stroke();
}

function setupWheel(){
  if(!wheelGames.length)return;
  spinCount=0;spinning=false;wheelRotation=0;
  const w=$("wheel");w.style.transition="none";w.style.transform="rotate(0deg)";
  drawWheel();
  $("spinBtn1").disabled=false;$("spinBtn1").textContent="SPIN GAME 1";
  $("spinBtn2").disabled=true;$("spinBtn2").textContent="SPIN GAME 2";$("spinBtn2").classList.remove("ready");
  $("spinStatus").textContent="Ready — spin Game 1";
  $("selectedGame").classList.add("hidden");$("selectedGame").innerHTML="";
}

function spinWheel(which){
  if(spinning||spinCount>=2||!wheelGames.length||currentGames.length<2)return;
  if(which===1&&spinCount!==0)return;if(which===2&&spinCount!==1)return;
  const targetGame=currentGames[which-1];if(!targetGame)return;
  spinning=true;$("spinBtn1").disabled=true;$("spinBtn2").disabled=true;
  $("spinStatus").textContent=which===1?"Spinning Game 1…":"Spinning Game 2…";

  let index=wheelGames.findIndex(g=>g.home===targetGame.home&&g.away===targetGame.away&&Math.abs(new Date(g.kick)-new Date(targetGame.kick))<120000);
  if(index<0)index=Math.min(which-1,wheelGames.length-1);

  const segment=360/wheelGames.length,targetAngle=-(index*segment+segment/2);
  const currentMod=((wheelRotation%360)+360)%360;
  const turns=6+Math.floor(Math.random()*3);
  const finalRotation=wheelRotation+turns*360+targetAngle-currentMod;
  const w=$("wheel");
  w.style.transition="transform 5.2s cubic-bezier(.08,.78,.16,1)";
  w.style.transform=`rotate(${finalRotation}deg)`;wheelRotation=finalRotation;

  setTimeout(()=>{
    spinCount++;
    $("selectedGame").innerHTML+=`<div style="margin:8px 0"><strong>Game ${which}: ${esc(targetGame.home)} v ${esc(targetGame.away)}</strong><small>${dateLabel(targetGame.kick)}</small></div>`;
    $("selectedGame").classList.remove("hidden");
    drawWheel();spinning=false;
    if(spinCount===1){
      $("spinBtn2").disabled=false;$("spinBtn2").classList.add("ready");
      $("spinStatus").textContent="Game 1 selected — now spin Game 2";
    }else{
      $("spinBtn2").textContent="BOTH GAMES SELECTED ✓";
      $("spinStatus").textContent="Both SUPER 2s games revealed";
    }
  },5400);
}

async function loadAllWeekendGames(weekStart){
  const base=new Date(weekStart+"T00:00:00"),out=[];
  for(let i=0;i<4;i++){
    const d=new Date(base);d.setDate(base.getDate()+i);
    const ds=d.toISOString().slice(0,10).replaceAll("-","");
    try{
      const res=await fetch(`https://site.api.espn.com/apis/site/v2/sports/soccer/eng.1/scoreboard?dates=${ds}`,{cache:"no-store"});
      if(!res.ok)continue;const data=await res.json();
      for(const e of data.events||[]){
        const c=(e.competitions||[])[0]||{},teams=c.competitors||[];if(teams.length!==2)continue;
        const home=teams.find(x=>x.homeAway==="home")||teams[0],away=teams.find(x=>x.homeAway==="away")||teams[1],kick=e.date||c.date;
        if(!kick)continue;const day=new Date(kick).getDay();if(![0,1,5,6].includes(day))continue;
        out.push({id:e.id,home:home.team?.displayName||"",away:away.team?.displayName||"",kick:new Date(kick).toISOString(),homeLogo:home.team?.logo||"",awayLogo:away.team?.logo||""});
      }
    }catch(e){console.warn("Fixture fetch failed",e)}
  }
  const seen=new Set();return out.filter(g=>g.home&&g.away&&g.id&&!seen.has(g.id)&&seen.add(g.id)).sort((a,b)=>new Date(a.kick)-new Date(b.kick));
}

async function loadWeek(){
  try{
    const r=await sb.from("weeks").select("*").order("week_start",{ascending:false}).limit(1).maybeSingle();
    if(r.error){$("games").innerHTML='<div class="msg">Database connection error: '+esc(r.error.message)+"</div>";$("spinStatus").textContent="Unable to load games";return}
    currentWeek=r.data;
    if(!currentWeek){$("games").innerHTML='<div class="msg">No games have been selected yet.</div>';return}
    currentGames=[
      {home:currentWeek.game1_home,away:currentWeek.game1_away,kick:currentWeek.game1_kickoff,n:1},
      {home:currentWeek.game2_home,away:currentWeek.game2_away,kick:currentWeek.game2_kickoff,n:2}
    ];
    wheelGames=await loadAllWeekendGames(currentWeek.week_start);
    const selected=currentGames.map(c=>wheelGames.find(g=>g.home===c.home&&g.away===c.away)||({...c,id:"selected-"+c.n,homeLogo:"",awayLogo:""}));
    const ids=new Set(selected.map(g=>g.id));
    wheelGames=[...selected,...wheelGames.filter(g=>!ids.has(g.id))].slice(0,10);
    $("wheelCount").textContent=wheelGames.length+" eligible fixtures";
    renderGames();setupWheel();await loadWinner();
  }catch(e){console.error(e);$("spinStatus").textContent="Unable to load wheel"}
}

function renderGames(){$("games").innerHTML=currentGames.map(g=>`<div class="game"><b>${esc(g.home)} v ${esc(g.away)}</b><div class="kick">${dateLabel(g.kick)}</div></div>`).join("")}

async function continuePlayer(){
  const name=$("name").value.trim(),pin=$("pin").value.trim();

  if(x.error){msg(x.error.message);$("continueBtn").disabled=false;return}
  if(x.data){showSubmitted(x.data);await showOthers();return}
  renderPredictionForm();$("predForm").classList.remove("hidden");$("continueBtn").classList.add("hidden");
}
